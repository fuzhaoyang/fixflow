# 告警功能实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: 按 superpowers:executing-plans 逐个任务执行。

**Goal:** 面板头部红色铃铛告警入口，弹窗内管理告警（增删改/启停/手动执行）、独立 SMTP 配置与执行历史；每日固定时间自动调用模型生成报表发邮件。

**Architecture:** 新增 `src/alerts.js`（调度+执行+历史，依赖注入便于测试），`server.js` 挂载并暴露 `/api/alerts*`，前端用铃铛+弹窗。零第三方依赖。

**Spec:** `docs/superpowers/specs/2026-09-23-alerts-design.md`

## Global Constraints
- 零依赖；`npm test`；告警存 `config.json` 的 `alerts`/`alertSmtp`，历史存 `alert-history.json`。
- 定时执行 `onConfirm` 自动拒绝；去重键 `id:当天`。
- 注释中文，不加多余注释。

```js
import { readFile, writeFile, mkdir } from "node:fs/promises";
import { dirname, join } from "node:path";
import { createSession } from "./core.js";
import { sendEmail } from "./ssl.js";
import { runCommand } from "./tools.js";
import { formatDateTime } from "./metrics.js";
import { dayKey } from "./usage.js";

export async function createAlertsManager({ config, configPath, exec = runCommand, sessionFactory, now = () => new Date() } = {}) {
  const historyPath = join(dirname(configPath), "alert-history.json");
  const readCfg = async () => { try { return JSON.parse(await readFile(configPath, "utf8")); } catch { return {}; } };
  const writeCfg = async (patch) => {
    const existing = await readCfg();
    await writeFile(configPath, JSON.stringify({ ...existing, ...patch }, null, 2));
  };
  let history = {};
  try { history = JSON.parse(await readFile(historyPath, "utf8")); } catch {}
  const persistHistory = async () => {
    await mkdir(dirname(historyPath), { recursive: true });
    await writeFile(historyPath, JSON.stringify(history, null, 2)).catch(() => {});
  };
  const ranToday = new Set();
  const makeSession = sessionFactory || ((cfg) => createSession({ config: cfg, onConfirm: async () => false, onEvent: () => {}, onUsage: () => {} }));

  const list = async () => (await readCfg()).alerts || [];
  const save = async (alerts) => writeCfg({ alerts });
  const smtp = async () => (await readCfg()).alertSmtp || {};
  const saveSmtp = async (s) => {
    const cur = await smtp();
    writeCfg({ alertSmtp: {
      host: s.host ?? cur.host ?? "", port: s.port ?? cur.port ?? "", user: s.user ?? cur.user ?? "",
      pass: s.pass ? s.pass : (cur.pass || ""), from: s.from ?? cur.from ?? "",
    } });
  };
  const historyFor = async (id) => history[id] || [];
  const addHistory = async (id, entry) => {
    const arr = (history[id] = history[id] || []);
    arr.unshift(entry);
    if (arr.length > 20) arr.length = 20;
    await persistHistory();
  };

  const run = async (alert) => {
    const entry = { at: formatDateTime(now()), ok: false, subject: alert.name, summary: "", error: "" };
    try {
      if (!alert.recipients?.length) throw new Error("未配置收件人");
      const session = makeSession(config);
      const report = await session.send(alert.instruction, []);
      entry.summary = String(report || "").slice(0, 500);
      const st = await smtp();
      const sent = await sendEmail({ ...st, to: alert.recipients.join(",") }, alert.name, String(report || ""), exec);
      if (!sent) throw new Error("邮件发送失败（未检测到 sendmail 或发送异常）");
      entry.ok = true;
    } catch (e) {
      entry.error = String(e?.message || e).slice(0, 500);
    }
    await addHistory(alert.id, entry);
    return entry;
  };

  const runNow = async (id) => {
    const alerts = await list();
    const a = alerts.find((x) => x.id === id);
    if (!a) return { ok: false, error: "告警不存在" };
    return run(a);
  };

  const tick = async () => {
    const alerts = await list();
    const n = now();
    const hm = `${String(n.getHours()).padStart(2, "0")}:${String(n.getMinutes()).padStart(2, "0")}`;
    const key = dayKey(n);
    for (const a of alerts) {
      if (!a.enabled || hm < (a.time || "")) continue;
      const k = `${a.id}:${key}`;
      if (ranToday.has(k)) continue;
      ranToday.add(k);
      await run(a).catch(() => {});
    }
  };

  return { list, save, smtp, saveSmtp, history: historyFor, runNow, tick };
}
```

Note: `saveSmtp`/`save` 未 await 是 bug——改为 `async` 内 `await writeCfg(...)`。

async function refreshAlerts() {
  try {
    const res = await fetch("api/alerts");
    const data = await res.json();
    document.getElementById("bell-dot").classList.toggle("on", (data.alerts || []).some((a) => a.enabled));
    document.getElementById("alert-host").value = data.smtp?.host || "";
    document.getElementById("alert-port").value = data.smtp?.port || "";
    document.getElementById("alert-user").value = data.smtp?.user || "";
    document.getElementById("alert-from").value = data.smtp?.from || "";
    const list = document.getElementById("alert-list");
    if (!data.alerts?.length) { list.innerHTML = `<pre style="color:var(--muted)">（暂无告警）</pre>`; }
    else {
      list.innerHTML = data.alerts.map((a) => `<div class="alert-row">
        <span class="tag">${a.enabled ? "启用" : "停用"}</span>
        <b>${escapeHtml(a.name)}</b>
        <span>${escapeHtml(a.time)}</span>
        <span style="color:var(--muted);font-size:12px">${escapeHtml((a.recipients || []).join(", "))}</span>
        <span class="proc-ops">
          <a href="javascript:void(0)" data-alert-run="${escapeHtml(a.id)}">立即执行</a>
          <a href="javascript:void(0)" data-alert-toggle="${escapeHtml(a.id)}" data-enabled="${a.enabled}">${a.enabled ? "停用" : "启用"}</a>
          <a href="javascript:void(0)" data-alert-edit="${escapeHtml(a.id)}">编辑</a>
          <a href="javascript:void(0)" data-alert-del="${escapeHtml(a.id)}" class="danger">删除</a>
        </span>
      </div>`).join("");
    }
    const hist = document.getElementById("alert-hist");
    const acc = [];
    for (const a of data.alerts || []) {
      const hs = (data.history?.[a.id] || []).map((h) => `<div class="alert-row"><span class="tag ${h.ok ? "ok" : "bad"}">${h.ok ? "成功" : "失败"}</span><b>${escapeHtml(h.subject)}</b><span style="color:var(--muted);font-size:12px">${escapeHtml(h.at)}</span><span style="color:var(--muted);font-size:12px;max-width:300px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">${escapeHtml(h.error || h.summary)}</span></div>`).join("");
      if (hs) acc.push(hs);
    }
    hist.innerHTML = acc.join("") || `<pre style="color:var(--muted)">（暂无执行记录）</pre>`;
  } catch {}
}

  const body = {
    name: document.getElementById("alert-name").value.trim(),
    time: document.getElementById("alert-time").value,
    instruction: document.getElementById("alert-instruction").value.trim(),
    recipients: document.getElementById("alert-recipients").value.split(/[,，\s]+/).filter(Boolean),
  };
  const errEl = document.getElementById("alert-form-err");
  if (!body.name || !body.time || !body.instruction || !body.recipients.length) { errEl.textContent = "名称/时间/指令/收件人必填"; return; }
  errEl.textContent = "";
  const res = await fetch("api/alerts", { method: alertEditId ? "PATCH" : "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(alertEditId ? { ...body, id: alertEditId } : body) });
  if (!res.ok) { errEl.textContent = ((await res.json()).error) || "保存失败"; return; }
  document.getElementById("alert-form").style.display = "none";
  refreshAlerts();
};

document.getElementById("alert-list").addEventListener("click", async (e) => {
  const a = e.target.closest("[data-alert-edit],[data-alert-del],[data-alert-run],[data-alert-toggle]");
  if (!a) return;
  const id = a.dataset.alertEdit || a.dataset.alertDel || a.dataset.alertRun || a.dataset.alertToggle;
  if (a.dataset.alertDel) {
    if (!confirm("确认删除该告警？")) return;
    await fetch("api/alerts", { method: "DELETE", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id }) });
    refreshAlerts();
  } else if (a.dataset.alertToggle) {
    await fetch("api/alerts", { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id, enabled: a.dataset.enabled === "false" }) });
    refreshAlerts();
  } else if (a.dataset.alertRun) {
    const res = await fetch("api/alerts/run", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id }) });
    showSaveMsg(res.ok ? "✅ 已执行" : "❌ " + ((await res.json()).error || "执行失败"));
    refreshAlerts();
  } else if (a.dataset.alertEdit) {
    alertEditId = id;
    const res = await fetch("api/alerts");
    const data = await res.json();
    const it = data.alerts.find((x) => x.id === id);
    document.getElementById("alert-form-title").textContent = "编辑告警";
    document.getElementById("alert-name").value = it.name;
    document.getElementById("alert-time").value = it.time;
    document.getElementById("alert-recipients").value = (it.recipients || []).join(", ");
    document.getElementById("alert-instruction").value = it.instruction;
    document.getElementById("alert-form-err").textContent = "";
    document.getElementById("alert-form").style.display = "";
  }
});

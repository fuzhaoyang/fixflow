# fixflow 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现零依赖的服务器运维 AI Agent fixflow，提供 CLI 交互会话与 Web 管理面板，接入 OpenAI 兼容大模型（默认 Kimi K2.7 Code），先检查后确认地诊断与修复服务器问题。

**Architecture:** 单一 Node ESM 进程，CLI 与 Web 共用 `src/core.js` 的 AgentLoop（会话历史 + function calling 工具调度）。只读工具自动执行，`apply_fix` 经用户确认后执行。Web 面板用原生 HTML/JS，HTTP 接口 + SSE 实时推送，`127.0.0.1` 无鉴权。

**Tech Stack:** Node.js ≥18（内置 fetch/readline/http/child_process）、ESM、零第三方依赖、测试用内置 `node:test`。

**Spec:** `docs/superpowers/specs/2026-09-16-fixflow-design.md`

## Global Constraints

- Node ≥18，`"type": "module"`，`"engines": { "node": ">=18" }`
- 零第三方依赖，仅用 Node 内置模块
- 所有对外解释、日志、界面文案使用中文
- 命令超时默认 30s；单条输出截断 50KB（`MAX_OUTPUT = 50 * 1024`）
- Web 监听 `127.0.0.1:8080`，无鉴权
- 默认模型 `kimi-k2.7-code`，默认 baseUrl `https://api.moonshot.cn/v1`
- 测试命令统一：`node --test test/`

// src/tools.js
import { exec } from "node:child_process";
import { readFile as fsReadFile } from "node:fs/promises";

const MAX_OUTPUT = 50 * 1024;
const truncate = (s) => (s.length > MAX_OUTPUT ? s.slice(0, MAX_OUTPUT) + "\n...[已截断]" : s);

export function runCommand(command, { timeout = 30000 } = {}) {
  return new Promise((resolve) => {
    exec(command, { timeout, maxBuffer: MAX_OUTPUT }, (err, stdout, stderr) => {
      resolve({
        stdout: truncate(stdout || ""),
        stderr: truncate(stderr || ""),
        exitCode: err ? (typeof err.code === "number" ? err.code : 1) : 0,
      });
    });
  });
}

export async function readFile(path) {
  try {
    return { content: truncate(await fsReadFile(path, "utf8")), error: null };
  } catch (e) {
    return { content: "", error: e.message };
  }
}

export async function applyFix(command, onConfirm) {
  const ok = await onConfirm(command);
  if (!ok) return { confirmed: false, note: "用户拒绝执行" };
  const r = await runCommand(command);
  return { confirmed: true, ...r };
}
```

// src/agent.js
export async function chat({ apiKey, baseUrl, model, messages, tools }) {
  const res = await fetch(`${baseUrl.replace(/\/$/, "")}/chat/completions`, {
    method: "POST",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${apiKey}` },
    body: JSON.stringify({ model, messages, tools, tool_choice: "auto" }),
  });
  if (!res.ok) throw new Error(`LLM API ${res.status}: ${await res.text()}`);
  const data = await res.json();
  const msg = data.choices?.[0]?.message || {};
  const toolCalls = (msg.tool_calls || []).map((tc) => ({
    id: tc.id,
    name: tc.function.name,
    args: JSON.parse(tc.function.arguments || "{}"),
  }));
  return { text: msg.content || "", toolCalls };
}
```

// src/core.js
import { chat as defaultChat } from "./agent.js";
import { runCommand, readFile, applyFix } from "./tools.js";

const TOOLS = [
  { type: "function", function: { name: "run_command", description: "执行 shell 命令检查服务器状态（只读诊断用）。", parameters: { type: "object", properties: { command: { type: "string", description: "要执行的命令" } }, required: ["command"] } } },
  { type: "function", function: { name: "read_file", description: "读取服务器上的文件内容（日志、配置等）。", parameters: { type: "object", properties: { path: { type: "string", description: "文件绝对路径" } }, required: ["path"] } } },
  { type: "function", function: { name: "apply_fix", description: "执行修复操作。调用前必须向用户说明原因，此命令需用户确认后才能执行。", parameters: { type: "object", properties: { command: { type: "string", description: "修复命令" }, reason: { type: "string", description: "为什么要执行这条命令" } }, required: ["command", "reason"] } } },
];

const HANDLERS = {
  run_command: ({ command }) => runCommand(command),
  read_file: ({ path }) => readFile(path),
  apply_fix: ({ command }, ctx) => applyFix(command, ctx.onConfirm),
};

export function createSession({ config, onConfirm = async () => false, onEvent, llmChat = defaultChat }) {
  const history = [];
  const ctx = { onConfirm };
  async function send(input) {
    history.push({ role: "user", content: input });
    for (;;) {
      const { text, toolCalls } = await llmChat({
        apiKey: config.apiKey, baseUrl: config.baseUrl, model: config.model,
        messages: history, tools: TOOLS,
      });
      if (text) {
        history.push({ role: "assistant", content: text });
        onEvent?.({ type: "text", text });
      }
      if (!toolCalls.length) return text || "";
      for (const call of toolCalls) {
        onEvent?.({ type: "tool", name: call.name, args: call.args });
        const handler = HANDLERS[call.name];
        const result = handler ? await handler(call.args, ctx) : { error: `未知工具 ${call.name}` };
        history.push({ role: "tool", tool_call_id: call.id, content: JSON.stringify(result) });
      }
    }
  }
  return { send, history };
}
```

// src/cli.js
import readline from "node:readline/promises";
import { stdin, stdout } from "node:process";
import { createSession } from "./core.js";

export function makeConfirm(rl) {
  return async (command) => {
    const answer = await rl.question(`\n[y/N] 执行: ${command}\n`);
    return /^y/i.test(answer.trim());
  };
}

export async function startFlow(config) {
  const rl = readline.createInterface({ input: stdin, output: stdout });
  const session = createSession({ config, onConfirm: makeConfirm(rl) });
  console.log("fixflow 服务器运维 Agent · 输入指令开始检查，Ctrl+C 退出");
  for (;;) {
    const input = await rl.question("> ");
    if (input.trim() === "exit" || input.trim() === "quit") break;
    if (!input.trim()) continue;
    const out = await session.send(input);
    if (out) console.log(`\n${out}\n`);
  }
  rl.close();
}
```

// src/metrics.js
import { runCommand } from "./tools.js";

const safe = async (fn) => {
  try { return await fn(); } catch { return { error: "N/A" }; }
};

export async function collectSystem() {
  const [uptime, mem, disk] = await Promise.all([
    runCommand("uptime"), runCommand("free -m"), runCommand("df -h"),
  ]);
  return {
    uptime: { ok: uptime.exitCode === 0, detail: uptime.stdout || uptime.stderr },
    memory: { ok: mem.exitCode === 0, detail: mem.stdout || mem.stderr },
    disk: { ok: disk.exitCode === 0, detail: disk.stdout || disk.stderr },
  };
}

export async function collectNetwork() {
  const ss = await runCommand("ss -tulnp");
  return { listeners: { ok: ss.exitCode === 0, detail: ss.stdout || ss.stderr } };
}

export async function collectNginx() {
  const [status, config] = await Promise.all([
    runCommand("systemctl is-active nginx"),
    runCommand("nginx -t 2>&1"),
  ]);
  return {
    status: { ok: status.exitCode === 0, detail: status.stdout.trim() || "未安装/未运行" },
    config: { ok: config.exitCode === 0, detail: config.stdout || config.stderr },
  };
}

export async function collectFirewall() {
  const [ufw, iptables] = await Promise.all([
    runCommand("ufw status"),
    runCommand("iptables -L -n --line-numbers | head -30"),
  ]);
  const isActive = ufw.stdout.includes("Status: active") || ufw.exitCode === 0 && ufw.stdout.includes("active");
  return {
    status: { ok: true, detail: isActive ? "ufw active" : (ufw.stdout.trim() || "ufw 不可用") },
    rules: { ok: iptables.exitCode === 0, detail: iptables.stdout || iptables.stderr },
  };
}

export async function collectServices({ services = [] }) {
  return Promise.all(services.map(async (s) => {
    try {
      const res = await fetch(s.url, { signal: AbortSignal.timeout(5000) });
      return { name: s.name, ok: res.status < 500, detail: `HTTP ${res.status}` };
    } catch (e) {
      return { name: s.name, ok: false, detail: e.message };
    }
  }));
}

export async function collectDatabases({ databases = [] }) {
  const probes = {
    postgres: "pg_isready",
    mysql: "mysqladmin ping",
    redis: "redis-cli ping",
  };
  return Promise.all(databases.map(async (d) => {
    const cmd = probes[d.type];
    if (!cmd) return { name: d.name, ok: false, detail: `未知数据库类型 ${d.type}` };
    const r = await runCommand(cmd);
    return { name: d.name, ok: r.exitCode === 0, detail: (r.stdout || r.stderr).trim() || "无输出" };
  }));
}

export async function collectDocker() {
  const r = await runCommand("docker ps -a --format '{{.Names}} | {{.Status}}'");
  return {
    ok: r.exitCode === 0,
    detail: r.stdout || r.stderr || "docker 不可用",
  };
}

export async function collectAll(config) {
  return {
    system: await safe(collectSystem),
    network: await safe(collectNetwork),
    nginx: await safe(collectNginx),
    firewall: await safe(collectFirewall),
    services: await safe(() => collectServices(config)),
    databases: await safe(() => collectDatabases(config)),
    docker: await safe(collectDocker),
  };
}

// src/web/server.js
import { createServer } from "node:http";
import { readFile } from "node:fs/promises";
import { fileURLToPath } from "node:url";
import { join } from "node:path";
import { createSession } from "../core.js";
import { collectAll } from "../metrics.js";

const STATIC = join(fileURLToPath(new URL(".", import.meta.url)), "static");

export async function startWeb(config = {}) {
  const clients = new Set();
  const waiters = new Map();
  let seq = 0;

  const broadcast = (event) => {
    const data = `data: ${JSON.stringify(event)}\n\n`;
    for (const res of clients) res.write(data);
  };

  const session = createSession({
    config,
    onEvent: broadcast,
    onConfirm: (command) => new Promise((resolve) => {
      const id = String(++seq);
      waiters.set(id, resolve);
      broadcast({ type: "confirm", id, command });
    }),
  });

  const server = createServer(async (req, res) => {
    const url = new URL(req.url, "http://127.0.0.1");
    try {
      if (url.pathname === "/api/metrics") {
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify(await collectAll(config)));
      } else if (url.pathname === "/api/chat" && req.method === "POST") {
        let body = "";
        for await (const chunk of req) body += chunk;
        const { input } = JSON.parse(body || "{}");
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end(JSON.stringify({ ok: true }));
        session.run(input).catch((e) => broadcast({ type: "error", text: String(e) }));
      } else if (url.pathname === "/api/confirm" && req.method === "POST") {
        let body = "";
        for await (const chunk of req) body += chunk;
        const { id, ok } = JSON.parse(body || "{}");
        const resolve = waiters.get(id);
        if (resolve) { waiters.delete(id); resolve(Boolean(ok)); }
        res.writeHead(200, { "Content-Type": "application/json" });
        res.end("{}");
      } else if (url.pathname === "/api/stream") {
        res.writeHead(200, { "Content-Type": "text/event-stream", "Cache-Control": "no-cache", Connection: "keep-alive" });
        res.write(": connected\n\n");
        clients.add(res);
        req.on("close", () => clients.delete(res));
      } else if (url.pathname === "/") {
        res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
        res.end(await readFile(join(STATIC, "index.html")));
      } else if (url.pathname === "/app.js") {
        res.writeHead(200, { "Content-Type": "text/javascript; charset=utf-8" });
        res.end(await readFile(join(STATIC, "app.js")));
      } else {
        res.writeHead(404); res.end("Not Found");
      }
    });
  });

  await new Promise((resolve) => server.listen(config.port || 8080, "127.0.0.1", resolve));
  const address = server.address();
  console.log(`fixflow 面板已启动：http://127.0.0.1:${address.port}`);
  return {
    server,
    close: () => new Promise((resolve) => server.close(resolve)),
  };
}

// src/web/static/app.js
const cards = document.getElementById("cards");
const log = document.getElementById("log");
const input = document.getElementById("input");
const conn = document.getElementById("conn");

const MODULES = [
  ["system", "系统"],
  ["network", "网络"],
  ["nginx", "Nginx"],
  ["firewall", "防火墙"],
  ["services", "Python 服务"],
  ["databases", "数据库"],
  ["docker", "Docker"],
];

function renderCards(data) {
  cards.innerHTML = "";
  for (const [key, label] of MODULES) {
    const m = data[key];
    const card = document.createElement("div");
    card.className = "card";
    const ok = !Array.isArray(m) ? m?.ok !== false : true;
    card.innerHTML = `<h2>${ok ? "✅" : "⚠️"} ${label}</h2><pre>${escapeHtml(detail(m))}</pre>`;
    cards.appendChild(card);
  }
}

function detail(m) {
  if (Array.isArray(m)) {
    if (!m.length) return "（未配置）";
    return m.map((x) => `${x.ok ? "✅" : "❌"} ${x.name}: ${x.detail ?? ""}`).join("\n");
  }
  if (!m) return "N/A";
  const parts = Object.entries(m).map(([k, v]) => `${k}: ${v?.detail ?? v?.ok ?? v ?? "N/A"}`);
  return parts.join("\n");
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
}

function addMsg(text, cls) {
  const div = document.createElement("div");
  div.className = "msg " + cls;
  div.textContent = text;
  log.appendChild(div);
  log.scrollTop = log.scrollHeight;
}

async function refreshMetrics() {
  try {
    const res = await fetch("/api/metrics");
    renderCards(await res.json());
  } catch {}
}

function connectStream() {
  const es = new EventSource("/api/stream");
  es.onopen = () => (conn.textContent = "已连接");
  es.onerror = () => (conn.textContent = "连接断开，重连中…");
  es.onmessage = (e) => {
    const ev = JSON.parse(e.data);
    if (ev.type === "text") addMsg(ev.text, "assistant");
    else if (ev.type === "tool") addMsg(`🔧 ${ev.name} ${JSON.stringify(ev.args)}`, "tool");
    else if (ev.type === "confirm") {
      addMsg(`⚠️ 需确认执行：${ev.command}`, "confirm");
      const row = document.createElement("div");
      row.className = "msg confirm";
      const yes = document.createElement("button");
      yes.className = "confirm-yes"; yes.textContent = "确认执行";
      yes.onclick = () => fetch("/api/confirm", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id: ev.id, ok: true }) });
      const no = document.createElement("button");
      no.className = "confirm-no"; no.textContent = "拒绝";
      no.onclick = () => fetch("/api/confirm", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ id: ev.id, ok: false }) });
      row.append(yes, no);
      log.appendChild(row);
      log.scrollTop = log.scrollHeight;
    } else if (ev.type === "error") addMsg(`❌ ${ev.text}`, "assistant");
  };
}

async function send() {
  const text = input.value.trim();
  if (!text) return;
  input.value = "";
  addLog(text, "user");
  await fetch("/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ input: text }) });
}

document.getElementById("send").onclick = send;
input.addEventListener("keydown", (e) => { if (e.key === "Enter") send(); });
refreshMetrics();
setInterval(refreshMetrics, 5000);
connectStream();
